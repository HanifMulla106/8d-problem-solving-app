<?php
class Database {
    private ?PDO $pdo = null;

    public function __construct(
        private string $host = DB_HOST,
        private string $dbName = DB_NAME,
        private string $user = DB_USER,
        private string $pass = DB_PASS
    ) {}

    public function connect(): PDO {
        if ($this->pdo instanceof PDO) return $this->pdo;

        $dsn = 'mysql:host=' . $this->host . ';dbname=' . $this->dbName . ';charset=utf8mb4';

        try {
            $this->pdo = new PDO($dsn, $this->user, $this->pass, [
                PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                PDO::ATTR_EMULATE_PREPARES => false,
            ]);
        } catch (PDOException $e) {
            die('Database connection failed: ' . $e->getMessage());
        }

        return $this->pdo;
    }

    public function getConnection(): PDO {
        return $this->connect();
    }
}

<?php
class Auth {
    private PDO $pdo;

    public function __construct(PDO $pdo) {
        $this->pdo = $pdo;

        if (session_status() === PHP_SESSION_NONE) {
            session_name(APP_SESSION_NAME);
            session_start();
        }
    }

    public function login(string $username, string $password): bool {
        $stmt = $this->pdo->prepare('SELECT * FROM users WHERE username = ? AND is_active = 1 LIMIT 1');
        $stmt->execute([$username]);
        $user = $stmt->fetch();

        if (!$user || !password_verify($password, $user['password_hash'])) {
            return false;
        }

        $_SESSION['user_id'] = (int)$user['id'];
        $_SESSION['user_role'] = $user['role'];
        $_SESSION['user_name'] = $user['full_name'];
        $_SESSION['user_email'] = $user['email'];

        return true;
    }

    public function isLoggedIn(): bool {
        return isset($_SESSION['user_id']);
    }

    public function currentUser(): ?array {
        if (!$this->isLoggedIn()) return null;

        $stmt = $this->pdo->prepare('SELECT * FROM users WHERE id = ? LIMIT 1');
        $stmt->execute([$_SESSION['user_id']]);
        return $stmt->fetch() ?: null;
    }

    public function getUserRole(): ?string {
        return $_SESSION['user_role'] ?? null;
    }

    public function hasRole(string $role): bool {
        return ($this->getUserRole() === $role);
    }

    public function canManageCases(): bool {
        $role = $this->getUserRole();
        return in_array($role, ['admin', 'manager', 'engineer']);
    }

    public function canApproveCases(): bool {
        $role = $this->getUserRole();
        return in_array($role, ['admin', 'approver']);
    }

    public function logout(): void {
        session_unset();
        session_destroy();
    }
}

<?php
class UserModel {
    public function __construct(private PDO $pdo) {}

    public function create(array $data): bool {
        $stmt = $this->pdo->prepare('
            INSERT INTO users (username, email, password_hash, full_name, role, is_active) 
            VALUES (?, ?, ?, ?, ?, 1)
        ');
        
        return $stmt->execute([
            $data['username'],
            $data['email'],
            password_hash($data['password'], PASSWORD_BCRYPT),
            $data['full_name'],
            $data['role'] ?? 'viewer',
        ]);
    }

    public function getById(int $id): ?array {
        $stmt = $this->pdo->prepare('SELECT * FROM users WHERE id = ? LIMIT 1');
        $stmt->execute([$id]);
        return $stmt->fetch() ?: null;
    }

    public function getAll(): array {
        $stmt = $this->pdo->query('SELECT * FROM users ORDER BY created_at DESC');
        return $stmt->fetchAll();
    }

    public function update(int $id, array $data): bool {
        $fields = [];
        $values = [];
        
        foreach ($data as $key => $value) {
            if (in_array($key, ['username', 'email', 'full_name', 'role', 'is_active'])) {
                $fields[] = "$key = ?";
                $values[] = $value;
            }
        }
        
        $values[] = $id;
        $stmt = $this->pdo->prepare('UPDATE users SET ' . implode(', ', $fields) . ' WHERE id = ?');
        return $stmt->execute($values);
    }

    public function delete(int $id): bool {
        $stmt = $this->pdo->prepare('DELETE FROM users WHERE id = ?');
        return $stmt->execute([$id]);
    }
}

class CaseModel {
    public function __construct(private PDO $pdo) {}

    public function create(array $data): int|false {
        $stmt = $this->pdo->prepare('
            INSERT INTO cases (title, description, severity, department, status, created_by, created_at) 
            VALUES (?, ?, ?, ?, "open", ?, NOW())
        ');
        
        $result = $stmt->execute([
            $data['title'],
            $data['description'],
            $data['severity'],
            $data['department'],
            $_SESSION['user_id'] ?? 1,
        ]);
        
        if ($result) {
            $caseId = (int)$this->pdo->lastInsertId();
            $this->createSteps($caseId);
            return $caseId;
        }
        
        return false;
    }

    private function createSteps(int $caseId): void {
        $steps = ['D0: Plan', 'D1: Team', 'D2: Problem', 'D3: Containment', 'D4: Root Cause', 'D5: Corrective', 'D6: Validate', 'D7: Prevent', 'D8: Closure'];
        
        $stmt = $this->pdo->prepare('INSERT INTO case_steps (case_id, step_name, step_order, status) VALUES (?, ?, ?, "open")');
        
        foreach ($steps as $index => $step) {
            $stmt->execute([$caseId, $step, $index]);
        }
    }

    public function getById(int $id): ?array {
        $stmt = $this->pdo->prepare('SELECT * FROM cases WHERE id = ? LIMIT 1');
        $stmt->execute([$id]);
        return $stmt->fetch() ?: null;
    }

    public function getAll(array $filters = []): array {
        $query = 'SELECT * FROM cases WHERE 1=1';
        $params = [];
        
        if (!empty($filters['status'])) {
            $query .= ' AND status = ?';
            $params[] = $filters['status'];
        }
        
        if (!empty($filters['severity'])) {
            $query .= ' AND severity = ?';
            $params[] = $filters['severity'];
        }
        
        if (!empty($filters['department'])) {
            $query .= ' AND department = ?';
            $params[] = $filters['department'];
        }
        
        $query .= ' ORDER BY created_at DESC LIMIT 100';
        
        $stmt = $this->pdo->prepare($query);
        $stmt->execute($params);
        return $stmt->fetchAll();
    }

    public function getRecent(int $limit = 10): array {
        $stmt = $this->pdo->prepare('SELECT * FROM cases ORDER BY created_at DESC LIMIT ?');
        $stmt->bindValue(1, $limit, PDO::PARAM_INT);
        $stmt->execute();
        return $stmt->fetchAll();
    }

    public function update(int $id, array $data): bool {
        $fields = [];
        $values = [];
        
        foreach ($data as $key => $value) {
            if (in_array($key, ['title', 'description', 'severity', 'department', 'status'])) {
                $fields[] = "$key = ?";
                $values[] = $value;
            }
        }
        
        $values[] = $id;
        $stmt = $this->pdo->prepare('UPDATE cases SET ' . implode(', ', $fields) . ' WHERE id = ?');
        return $stmt->execute($values);
    }

    public function delete(int $id): bool {
        $this->pdo->prepare('DELETE FROM case_steps WHERE case_id = ?')->execute([$id]);
        $this->pdo->prepare('DELETE FROM actions WHERE case_id = ?')->execute([$id]);
        $stmt = $this->pdo->prepare('DELETE FROM cases WHERE id = ?');
        return $stmt->execute([$id]);
    }

    public function getSteps(int $caseId): array {
        $stmt = $this->pdo->prepare('SELECT * FROM case_steps WHERE case_id = ? ORDER BY step_order ASC');
        $stmt->execute([$caseId]);
        return $stmt->fetchAll();
    }

    public function updateStep(int $stepId, array $data): bool {
        $stmt = $this->pdo->prepare('UPDATE case_steps SET description = ?, status = ? WHERE id = ?');
        return $stmt->execute([$data['description'], $data['status'], $stepId]);
    }

    public function getStats(): array {
        $total = $this->pdo->query('SELECT COUNT(*) as count FROM cases')->fetch()['count'];
        $open = $this->pdo->query('SELECT COUNT(*) as count FROM cases WHERE status = "open"')->fetch()['count'];
        $critical = $this->pdo->query('SELECT COUNT(*) as count FROM cases WHERE severity = "critical"')->fetch()['count'];
        
        return ['total' => $total, 'open' => $open, 'critical' => $critical];
    }
}

class ActionModel {
    public function __construct(private PDO $pdo) {}

    public function create(array $data): int|false {
        $stmt = $this->pdo->prepare('
            INSERT INTO actions (case_id, step_id, title, description, assigned_to, status, priority, due_date, created_at) 
            VALUES (?, ?, ?, ?, ?, "open", ?, ?, NOW())
        ');
        
        $result = $stmt->execute([
            $data['case_id'],
            $data['step_id'],
            $data['title'],
            $data['description'],
            $data['assigned_to'],
            $data['priority'] ?? 'medium',
            $data['due_date'],
        ]);
        
        return $result ? (int)$this->pdo->lastInsertId() : false;
    }

    public function getByCaseId(int $caseId): array {
        $stmt = $this->pdo->prepare('SELECT * FROM actions WHERE case_id = ? ORDER BY created_at DESC');
        $stmt->execute([$caseId]);
        return $stmt->fetchAll();
    }

    public function update(int $id, array $data): bool {
        $stmt = $this->pdo->prepare('UPDATE actions SET status = ?, completed_at = ? WHERE id = ?');
        return $stmt->execute([
            $data['status'],
            $data['status'] === 'completed' ? date('Y-m-d H:i:s') : null,
            $id,
        ]);
    }

    public function delete(int $id): bool {
        $stmt = $this->pdo->prepare('DELETE FROM actions WHERE id = ?');
        return $stmt->execute([$id]);
    }
}

class DashboardModel {
    public function __construct(private PDO $pdo) {}

    public function getMetrics(): array {
        $caseModel = new CaseModel($this->pdo);
        $stats = $caseModel->getStats();
        
        $overdue = $this->pdo->query('SELECT COUNT(*) as count FROM actions WHERE status = "open" AND due_date < NOW()')->fetch()['count'];
        
        return [
            'total_cases' => $stats['total'],
            'open_cases' => $stats['open'],
            'critical_issues' => $stats['critical'],
            'overdue_actions' => $overdue,
        ];
    }

    public function getRecentCases(int $limit = 10): array {
        $caseModel = new CaseModel($this->pdo);
        return $caseModel->getRecent($limit);
    }

    public function getRecentActions(int $limit = 5): array {
        $stmt = $this->pdo->prepare('
            SELECT a.*, c.title as case_title, u.full_name 
            FROM actions a 
            JOIN cases c ON a.case_id = c.id 
            LEFT JOIN users u ON a.assigned_to = u.id 
            ORDER BY a.created_at DESC 
            LIMIT ?
        ');
        $stmt->bindValue(1, $limit, PDO::PARAM_INT);
        $stmt->execute();
        return $stmt->fetchAll();
    }
}

class CaseController {
    public function __construct(private PDO $pdo, private Auth $auth) {}

    public function index(): void {
        if (!$this->auth->isLoggedIn()) {
            redirect('/login');
        }

        $filters = [];
        if (!empty($_GET['status'])) $filters['status'] = $_GET['status'];
        if (!empty($_GET['severity'])) $filters['severity'] = $_GET['severity'];
        if (!empty($_GET['department'])) $filters['department'] = $_GET['department'];

        $caseModel = new CaseModel($this->pdo);
        $cases = $caseModel->getAll($filters);
        $user = $this->auth->currentUser();

        render('cases/index.php', [
            'cases' => $cases,
            'filters' => $filters,
            'user' => $user,
            'flash' => getFlash(),
        ]);
    }

    public function create(): void {
        if (!$this->auth->isLoggedIn() || !$this->auth->canManageCases()) {
            redirect('/login');
        }

        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $data = [
                'title' => $_POST['title'] ?? '',
                'description' => $_POST['description'] ?? '',
                'severity' => $_POST['severity'] ?? 'medium',
                'department' => $_POST['department'] ?? '',
            ];

            if (empty($data['title']) || empty($data['description'])) {
                flash('error', 'Title and description are required');
                redirect('/cases/create');
            }

            $caseModel = new CaseModel($this->pdo);
            $caseId = $caseModel->create($data);

            if ($caseId) {
                flash('success', 'Case created successfully');
                redirect('/cases/' . $caseId);
            } else {
                flash('error', 'Failed to create case');
                redirect('/cases/create');
            }
        }

        $user = $this->auth->currentUser();
        render('cases/create.php', ['user' => $user, 'flash' => getFlash()]);
    }

    public function view(int $id): void {
        if (!$this->auth->isLoggedIn()) {
            redirect('/login');
        }

        $caseModel = new CaseModel($this->pdo);
        $case = $caseModel->getById($id);

        if (!$case) {
            flash('error', 'Case not found');
            redirect('/cases');
        }

        $steps = $caseModel->getSteps($id);
        $actionModel = new ActionModel($this->pdo);
        $actions = $actionModel->getByCaseId($id);
        $user = $this->auth->currentUser();

        render('cases/view.php', [
            'case' => $case,
            'steps' => $steps,
            'actions' => $actions,
            'user' => $user,
            'flash' => getFlash(),
        ]);
    }

    public function update(int $id): void {
        if (!$this->auth->isLoggedIn() || !$this->auth->canManageCases()) {
            redirect('/login');
        }

        if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
            redirect('/cases/' . $id);
        }

        $data = [
            'status' => $_POST['status'] ?? 'open',
            'severity' => $_POST['severity'] ?? 'medium',
        ];

        $caseModel = new CaseModel($this->pdo);
        if ($caseModel->update($id, $data)) {
            flash('success', 'Case updated successfully');
        } else {
            flash('error', 'Failed to update case');
        }

        redirect('/cases/' . $id);
    }
}

CREATE DATABASE IF NOT EXISTS eightd_db;
USE eightd_db;

CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100),
    role ENUM('admin', 'manager', 'engineer', 'approver', 'viewer') DEFAULT 'viewer',
    is_active BOOLEAN DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_username (username),
    INDEX idx_email (email)
);

CREATE TABLE cases (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    description TEXT,
    severity ENUM('low', 'medium', 'high', 'critical') DEFAULT 'medium',
    department VARCHAR(100),
    status ENUM('open', 'in_progress', 'completed', 'closed') DEFAULT 'open',
    created_by INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (created_by) REFERENCES users(id),
    INDEX idx_status (status),
    INDEX idx_severity (severity),
    INDEX idx_created_at (created_at)
);

CREATE TABLE case_steps (
    id INT PRIMARY KEY AUTO_INCREMENT,
    case_id INT NOT NULL,
    step_name VARCHAR(100),
    step_order INT,
    description TEXT,
    status ENUM('open', 'in_progress', 'completed') DEFAULT 'open',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (case_id) REFERENCES cases(id) ON DELETE CASCADE,
    INDEX idx_case_id (case_id)
);

CREATE TABLE actions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    case_id INT NOT NULL,
    step_id INT,
    title VARCHAR(255) NOT NULL,
    description TEXT,
    assigned_to INT,
    status ENUM('open', 'in_progress', 'completed') DEFAULT 'open',
    priority ENUM('low', 'medium', 'high') DEFAULT 'medium',
    due_date DATE,
    completed_at DATETIME,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (case_id) REFERENCES cases(id) ON DELETE CASCADE,
    FOREIGN KEY (step_id) REFERENCES case_steps(id),
    FOREIGN KEY (assigned_to) REFERENCES users(id),
    INDEX idx_case_id (case_id),
    INDEX idx_assigned_to (assigned_to),
    INDEX idx_status (status)
);

INSTALLATION INSTRUCTIONS
================================================================================

STEP 1: Create Database
1. Open MySQL command line or phpMyAdmin
2. Run the schema.sql file:
   mysql -u root -p < db/schema.sql
3. Verify database created:
   mysql -u root -p -e "SHOW DATABASES LIKE 'eightd_db';"

STEP 2: Create Folder Structure
1. Create the folders exactly as shown in FOLDER STRUCTURE section above
2. Create each .php file with its code

STEP 3: Update Configuration
1. Edit config/database.php
2. Change DB_PASS if your MySQL password is not empty
3. Test connection by running:
   php -r "include 'config/database.php'; echo 'Config loaded';"

STEP 4: Start PHP Server
cd public
php -S localhost:8000

STEP 5: Setup Admin Account
1. Open browser: http://localhost:8000/setup.php
2. Fill in admin credentials
3. Click "Create Admin"
4. You'll be redirected to login

STEP 6: Login & Test
1. Go to: http://localhost:8000/login
2. Login with admin credentials
3. Create a test case
4. Verify all features work

TROUBLESHOOTING
- If 404 errors: Check .htaccess is in root folder
- If blank page: Check PHP error log
- If database error: Verify database credentials in config/database.php
- If "Class not found": Verify all class files are in src/ folder
